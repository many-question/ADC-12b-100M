Ideal switch model (top A high_scale=.9; other devices default1):
// Private SFDR group attribution only; not a manufacturable device model.
`include "constants.vams"
`include "disciplines.vams"
module cdac_diag_switch(A,B,G,R,H,X);
  inout A,B,G,R,H,X;
  electrical A,B,G,R,H,X;
  parameter real ron=100 from (0:inf);
  parameter real roff=1e12 from (0:inf);
  parameter real band=0.1 from (0:1);
  parameter integer active_high=1 from [0:1];
  parameter integer bootstrap=0 from [0:1];
  parameter real high_scale=1;
  real level,u,x,a,g;
  analog begin
    level=high_scale*V(H,R)+bootstrap*V(X,R);
    level=sqrt(level*level+1e-12);
    u=V(G,R)/level;
    if (active_high==0) u=1-u;
    x=(u-(0.5-band/2))/band;
    if (x<=0) a=0;
    else if (x>=1) a=1;
    else a=x*x*(3-2*x);
    g=1/roff+(1/ron-1/roff)*a;
    I(A,B) <+ g*V(A,B);
  end
endmodule
